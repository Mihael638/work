-- 4. Создание таблицы "Специальность"
CREATE TABLE "Специальность" (
    "Номер" VARCHAR(10) PRIMARY KEY,
    "Направление" VARCHAR(255) NOT NULL,
    "Шифр" VARCHAR(10) NOT NULL,
    
    -- Связь с таблицей "Кафедра"
    CONSTRAINT fk_specialnost_kafedra 
        FOREIGN KEY ("Шифр") 
        REFERENCES "Кафедра" ("Шифр")
);

-- 5. Создание таблицы "Дисциплина"

    -- Связь с таблицей "Кафедра" (Исполнитель - это шифр кафедры)
    CONSTRAINT fk_disciplina_kafedra 
        FOREIGN KEY ("Исполнитель") 
        REFERENCES "Кафедра" ("Шифр")
);

-- 6. Создание таблицы "Заявка"
-- Связующая таблица между "Специальность" и "Дисциплина"
CREATE TABLE "Заявка" (
    "Номер" VARCHAR(10) NOT NULL,
    "Код" INT NOT NULL,
    
    -- Составной первичный ключ (одна специальность не может дважды запросить одну дисциплину)
    PRIMARY KEY ("Номер", "Код"),
    
    -- Внешние ключи
    CONSTRAINT fk_zayavka_specialnost 
        FOREIGN KEY ("Номер") REFERENCES "Специальность" ("Номер"),
    CONSTRAINT fk_zayavka_disciplina 
        FOREIGN KEY ("Код") REFERENCES "Дисциплина" ("Код")
);

-- 11. Создание таблицы "Экзамен"
CREATE TABLE "Экзамен" (
    "Дата" DATE NOT NULL,
    "Код" INT NOT NULL,
    "Рег_номер" INT NOT NULL,
    "Таб_номер" INT NOT NULL,
    "Аудитория" VARCHAR(10) NOT NULL,
    "Оценка" INT NOT NULL,
    
    -- Составной первичный ключ
    PRIMARY KEY ("Дата", "Код", "Рег_номер", "Таб_номер"),
    
    -- Внешние ключи (связи с другими таблицами)
    CONSTRAINT fk_exam_disciplina 
        FOREIGN KEY ("Код") REFERENCES "Дисциплина" ("Код"),
        
    CONSTRAINT fk_exam_student 
        FOREIGN KEY ("Рег_номер") REFERENCES "Студент" ("Рег_номер"),
        
    CONSTRAINT fk_exam_sotrudnik 
        FOREIGN KEY ("Таб_номер") REFERENCES "Сотрудник" ("Таб_номер")
);
